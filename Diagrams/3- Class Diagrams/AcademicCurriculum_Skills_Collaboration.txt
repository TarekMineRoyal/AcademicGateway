    class Major {
        +Guid Id
        +string Name
        +IReadOnlyCollection~Specialty~ Specialties
        +IReadOnlyCollection~StudentMajor~ StudentMajors
        +UpdateName(string newName)
        +AddSpecialty(string name)
        +RemoveSpecialty(Guid specialtyId)
    }

    class Specialty {
        +Guid Id
        +string Name
        +Guid MajorId
        +IReadOnlyCollection~StudentSpecialty~ StudentSpecialties
        +UpdateName(string newName)
    }

    Major "1" *-- "*" Specialty : contains
