    Majors {

    ProjectTemplates {
        uuid id PK
        uuid provider_id FK
        uuid major_id FK "nullable"

        string description
        string status
        string reviewer_feedback "nullable"

    }

    GlobalTasks {
        uuid id PK
        uuid global_milestone_id FK

        string description
        decimal weight
        string required_deliverable_type
    }

    TemplateMilestoneDependencies {
        uuid predecessor_id PK, FK
